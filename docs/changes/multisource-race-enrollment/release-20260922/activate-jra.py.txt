def sha(raw):return hashlib.sha256(raw).hexdigest()
def blob(path):
 fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW)
 try:
  s=os.fstat(fd)
  if not stat.S_ISREG(s.st_mode) or s.st_size > 8*1024*1024 or s.st_uid!=os.getuid() or s.st_mode & 0o077:raise ValueError('private regular file required')
  return os.read(fd,8*1024*1024)
 finally:os.close(fd)
def save_blob(path,raw):
 if path.exists():
  if blob(path)!=raw:raise ValueError('immutable evidence differs')
  return
 fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
 try:
  with os.fdopen(fd,'wb') as f:f.write(raw);f.flush();os.fsync(f.fileno())
 except BaseException:raise

def replace_bound(path,raw,allowed):
 if sha(blob(path)) not in allowed:raise ValueError('configuration CAS drift')
 tmp=path.with_name(path.name+'.pr212-'+secrets.token_hex(6))
 save_blob(tmp,raw)
 os.replace(tmp,path)
 fd=os.open(path.parent,os.O_RDONLY|os.O_DIRECTORY)
 try:os.fsync(fd)
 finally:os.close(fd)

def change_env(raw):
 lines=raw.decode().splitlines();values={};counts={}
 for line in lines:
  if '=' in line and not line.startswith('#'):
   key,val=line.split('=',1);values[key]=val;counts[key]=counts.get(key,0)+1
 if any(n!=1 for n in counts.values()):raise ValueError('duplicate env key')
 providers=values['RACE_DATA_SYNC_ENABLED_PROVIDERS'].split(',')
 if set(providers)!={'horse_racing_nation','sporting_life','the_racing_api','zeturf'}:raise ValueError('unexpected original providers')
 changes={**CHANGES,'RACE_DATA_SYNC_ENABLED_PROVIDERS':','.join([*providers,'jra'])}
 result=[];seen=set()
 for line in lines:
  key=line.split('=',1)[0]
  if key in changes:line=key+'='+changes[key];seen.add(key)
  result.append(line)
 result.extend(key+'='+value for key,value in changes.items() if key not in seen)
 return ('\n'.join(result)+'\n').encode(),changes

def atomic_json(path,value):
 raw=c.encoded(value)
 if path.exists():replace_bound(path,raw,{sha(blob(path))})
 else:
  save_blob(path,raw)
  parent=os.open(path.parent,os.O_RDONLY|os.O_DIRECTORY)
  try:os.fsync(parent)
  finally:os.close(parent)
